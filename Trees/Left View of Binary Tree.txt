class Solution {

    ArrayList<Integer> ans = new ArrayList<>();

    void solve(Node root, int level) {

        if (root == null)
            return;

        if (level == ans.size()) {
            ans.add(root.data);
        }

        solve(root.left, level + 1);
        solve(root.right, level + 1);
    }

    ArrayList<Integer> leftView(Node root) {

        solve(root, 0);

        return ans;
    }
}
